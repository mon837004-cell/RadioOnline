        const seen = new Set();
        return (Array.isArray(list) ? list : [])
            .filter(item => item && typeof item === 'object' && (item.url_resolved || item.url))
            .map(normalizeStation)
            .filter(item => {
                if (!item.id || seen.has(item.id)) return false;
                seen.add(item.id);
                return true;
            });

    function loadGlobalFavorites() {
        try {
            let data = GM_getValue(STORAGE.favorites, null);
            if (data) {
                const parsed = typeof data === 'string' ? JSON.parse(data) : data;
                if (Array.isArray(parsed) && parsed.length > 0) return sanitizeFavorites(parsed);
            }

            const localData = localStorage.getItem(STORAGE.favorites);
            if (localData) {
                const parsed = JSON.parse(localData);
                if (Array.isArray(parsed) && parsed.length > 0) {
                    const clean = sanitizeFavorites(parsed);
                    GM_setValue(STORAGE.favorites, JSON.stringify(clean));
                    return clean;
                }
            }
            return [];
        } catch (_) {
            return [];
        }
    }

            "'": '&#39;'

                return await response.json();
            } catch (error) {
                lastError = error;
            } finally {
                clearTimeout(timer);
            }
        }
        throw lastError || new Error('No se pudo conectar al catálogo');

    async function loadStations(reset = false) {
        if (showingFavorites) {
            renderFavorites();
            return;
        }

        if (loading && !reset) return;

        if (reset) {
            offset = 0;
            currentQuery = getFilters();
            requestNumber++;
            results.innerHTML = '';
        }

        const myRequest = requestNumber;
        loading = true;

        if (reset) setStatus('Buscando emisoras…');

        try {
            const filters = currentQuery;
            const params = new URLSearchParams();
            params.set('limit', '50');
            params.set('offset', String(offset));
            params.set('hidebroken', 'true');
            params.set('order', 'clickcount');
            params.set('reverse', 'true');

            if (filters.name) params.set('name', filters.name);
            if (filters.country) params.set('country', filters.country);
            if (filters.genre) params.set('tag', filters.genre.toLowerCase());

            const data = await apiFetch('/stations/search?' + params.toString());

            if (myRequest !== requestNumber || !Array.isArray(data)) return;

            const stations = data
                .filter(s => s && (s.url_resolved || s.url) && s.stationuuid)
                .map(normalizeStation);

            offset += data.length;
            renderStations(stations, reset, data.length >= 50);

            if (reset && stations.length === 0) {
                setStatus('No se han encontrado emisoras. Prueba con otro país, género o búsqueda.');
            } else if (reset && stations.length > 0) {
                const count = document.createElement('div');
                count.className = 'rmw-status';
                count.textContent = 'Mostrando emisoras encontradas. Puedes cargar más al final.';
                results.prepend(count);
            }
        } catch (_) {
            if (myRequest === requestNumber) {
                setStatus('No se pudo conectar al directorio. Comprueba la conexión y pulsa «Actualizar».');
            }
        } finally {
            if (myRequest === requestNumber) loading = false;
        }
    }

              : `<div class="rmw-logo rmw-logo-placeholder">📻</div>`

        `;

        card.querySelector('[data-favorite]').addEventListener('click', () => {
            toggleFavorite(station);
            const button = card.querySelector('[data-favorite]');
            const nowFavorite = isFavorite(station);
            button.classList.toggle('is-fav', nowFavorite);
            button.textContent = nowFavorite ? '★' : '☆';
            if (showingFavorites) {
                const top = results.scrollTop;
                renderFavorites();
                results.scrollTop = top;
            }
        });

        return card;
    }

    function renderStations(stations, reset, hasMore = stations.length >= 50) {
        if (reset) results.innerHTML = '';
        stations.forEach(station => results.appendChild(stationCard(station)));

        const oldMore = $('#rmw-more');
        if (oldMore) oldMore.remove();

        if (hasMore) {
            const more = document.createElement('button');
            more.id = 'rmw-more';
            more.className = 'rmw-more';
            more.textContent = 'Cargar más emisoras ↓';
            more.addEventListener('click', () => {
                more.remove();
                loadStations(false);
            });
            results.appendChild(more);
        }
    }

    function renderFavorites() {
        results.innerHTML = '';

        const manager = document.createElement('div');
        manager.id = 'rmw-fav-manager';
        manager.innerHTML = `
          <div class="rmw-fav-info">⭐ Guardadas: ${favorites.length} emisoras</div>
          <div class="rmw-fav-btns">
            <button class="rmw-mgr-btn" id="rmw-export-file" title="Descargar archivo JSON con tu lista">📥 Exportar Favoritos</button>
            <button class="rmw-mgr-btn" id="rmw-import-file" title="Cargar archivo JSON desde tu equipo">📤 Importar Favoritos</button>
          </div>
        `;

        results.appendChild(manager);

        manager.querySelector('#rmw-export-file').addEventListener('click', exportFavoritesFile);
        manager.querySelector('#rmw-import-file').addEventListener('click', () => fileInput.click());

        if (!favorites.length) {
            const emptyMsg = document.createElement('div');
            emptyMsg.className = 'rmw-status';
            emptyMsg.textContent = 'No tienes emisoras guardadas. Marca ★ en cualquier radio o usa "Importar Favoritos" para restaurar una copia de seguridad.';
            results.appendChild(emptyMsg);
            return;
        }

        const query = searchInput.value.trim().toLowerCase();
        const country = countrySelect.value;
        const genre = genreSelect.value === 'Todos los géneros' ? '' : genreSelect.value.toLowerCase();

        const list = favorites.filter(station => {
            const matchesQuery = !query || (station.name + ' ' + station.country + ' ' + station.tags).toLowerCase().includes(query);
            const matchesCountry = !country || station.country === country;
            const matchesGenre = !genre || (station.tags || '').toLowerCase().includes(genre);
            return matchesQuery && matchesCountry && matchesGenre;
        });

        if (!list.length) {
            const noMatchMsg = document.createElement('div');
            noMatchMsg.className = 'rmw-status';
            noMatchMsg.textContent = 'No hay favoritos que coincidan con estos filtros.';
            results.appendChild(noMatchMsg);
            return;
        }

        list.forEach(station => results.appendChild(stationCard(station)));
    }

    function toggleFavorite(station) {
        const id = makeId(station);
        if (isFavorite(station)) {
            favorites = favorites.filter(f => f.id !== id);
        } else {
            favorites.push({
                id,
                name: station.name,
                url: station.url,
                country: station.country,
                countrycode: station.countrycode,
                language: station.language,
                tags: station.tags,
                favicon: station.favicon,
                codec: station.codec,
                bitrate: station.bitrate,
                homepage: station.homepage
            });
        }
        saveFavorites();
    }

    // ---------------------- GESTIÓN DE ARCHIVOS (IMPORT / EXPORT) ----------------------

    function exportFavoritesFile() {
        if (!favorites.length) {
            showToast('No tienes ningún favorito para exportar.');
            return;
        }
        const dataStr = 'data:text/json;charset=utf-8,' + encodeURIComponent(JSON.stringify(favorites, null, 2));
        const downloadAnchor = document.createElement('a');
        downloadAnchor.setAttribute('href', dataStr);
        downloadAnchor.setAttribute('download', 'mis_emisoras_favoritas.json');
        document.body.appendChild(downloadAnchor);
        downloadAnchor.click();
        downloadAnchor.remove();
    }

    fileInput.addEventListener('change', event => {
        const file = event.target.files[0];
        if (!file) return;

        const reader = new FileReader();
        reader.onload = e => {
            try {
                const imported = sanitizeFavorites(JSON.parse(e.target.result));
                if (imported.length > 0) {
                    favorites = imported;
                    saveFavorites();
                    showToast(`¡Éxito! Se han cargado ${imported.length} emisoras desde el archivo.`);
                    if (showingFavorites) renderFavorites();
                } else {
                    showToast('El archivo JSON no contiene una lista válida de emisoras.');
                }
            } catch (_) {
                showToast('Error al leer el archivo. Asegúrate de que sea un JSON válido.');
            }
            fileInput.value = '';
        };
        reader.readAsText(file);
    });

    // ---------------------- REPRODUCTOR & AUTORREANUDACIÓN ----------------------

    async function playStation(station) {
        userPaused = false;
        const normalized = normalizeStation(station);
        playingStation = normalized;
        updateMediaSession(normalized);
        mini.classList.add('rmw-visible');
        $('#rmw-mininame').textContent = normalized.name;
        $('#rmw-ministatus').textContent = 'Conectando…';
        $('#rmw-playpause').textContent = 'Ⅱ';

        if (loadedUrl !== normalized.url || !audio.src) {
            audio.pause();
            audio.src = normalized.url;
            loadedUrl = normalized.url;
            audio.load();
        }

        try {
            await audio.play();
            $('#rmw-ministatus').textContent = normalized.country + ' · En reproducción';
        } catch (_) {
            $('#rmw-ministatus').textContent = 'No se pudo reproducir. Prueba otra emisora.';
            $('#rmw-playpause').textContent = '▶';
        }

        refreshPlayButtons();
    }

    function refreshPlayButtons() {
        document.querySelectorAll('#rmw-modal [data-play]').forEach(button => {
            const btnId = button.getAttribute('data-play');
            const active = playingStation && playingStation.id === btnId;
            button.textContent = active && !audio.paused ? 'Ⅱ' : '▶';
            button.closest('.rmw-station')?.classList.toggle('rmw-playing', Boolean(active));
        });
    }

    $('#rmw-playpause').addEventListener('click', async () => {
        if (!audio.src) { openModal(); return; }
        if (audio.paused) {
            userPaused = false;
            try { await audio.play(); } catch (_) { $('#rmw-ministatus').textContent = 'No se pudo reanudar la emisora.'; }
        } else {
            userPaused = true;
            audio.pause();
        }
    });

    $('#rmw-volume').value = String(audio.volume);
    $('#rmw-volume').addEventListener('input', event => {
        audio.volume = Number(event.target.value);
        GM_setValue(STORAGE.volume, String(audio.volume));
    });

    async function tryAutoResume() {
        if (userPaused) { disarmAutoResume(); return; }
        if (!audio.paused || !audio.src) return;
        try {
            await audio.play();
            disarmAutoResume();
        } catch (_) {}
    }

    function onVisibilityResume() {
        if (document.visibilityState === 'visible') tryAutoResume();
    }

    function armAutoResume() {
        if (autoResumeArmed) return;
        autoResumeArmed = true;
        window.addEventListener('focus', tryAutoResume);
        window.addEventListener('click', tryAutoResume);
        document.addEventListener('visibilitychange', onVisibilityResume);
    }

    function disarmAutoResume() {
        if (!autoResumeArmed) return;
        autoResumeArmed = false;
        window.removeEventListener('focus', tryAutoResume);
        window.removeEventListener('click', tryAutoResume);
        document.removeEventListener('visibilitychange', onVisibilityResume);
    }

    audio.addEventListener('playing', () => {
        disarmAutoResume();
        $('#rmw-ministatus').textContent = (playingStation?.country || '') + ' · En reproducción';
        $('#rmw-playpause').textContent = 'Ⅱ';
        refreshPlayButtons();
    });

    audio.addEventListener('pause', () => {
        if (audio.src) {
            $('#rmw-ministatus').textContent = 'En pausa';
            $('#rmw-playpause').textContent = '▶';
        }
        refreshPlayButtons();

        if (!userPaused) armAutoResume();
    });

    audio.addEventListener('waiting', () => { $('#rmw-ministatus').textContent = 'Cargando audio…'; });
    audio.addEventListener('error', () => {
        $('#rmw-ministatus').textContent = 'Error de streaming. Prueba otra emisora.';
        $('#rmw-playpause').textContent = '▶';
        refreshPlayButtons();
    });

    // ---------------------- MEDIA SESSION ----------------------

    const hasMediaSession = 'mediaSession' in navigator;

    function updateMediaSession(station) {
        if (!hasMediaSession) return;
        try {
            navigator.mediaSession.metadata = new MediaMetadata({
                title: station.name,
                artist: station.country || 'Radio Forocochera',
                album: 'Radio Forocochera',
                artwork: ICON_URL ? [{ src: ICON_URL, sizes: '447x440', type: 'image/png' }] : []
            });
        } catch (_) {}
    }

    if (hasMediaSession) {
        const setHandler = (action, handler) => {
            try { navigator.mediaSession.setActionHandler(action, handler); } catch (_) {}
        };
        const pauseFromSystem = () => { userPaused = true; audio.pause(); };

        setHandler('play', async () => {
            if (!audio.src) return;
            userPaused = false;
            try { await audio.play(); } catch (_) {}
        });
        setHandler('pause', pauseFromSystem);
        setHandler('stop', pauseFromSystem);

        audio.addEventListener('playing', () => { navigator.mediaSession.playbackState = 'playing'; });
        audio.addEventListener('pause', () => { navigator.mediaSession.playbackState = 'paused'; });
    }

    // ---------------------- TEMPORIZADOR DE APAGADO ----------------------

    const TIMER_OPTIONS = [15, 30, 45, 60, 90];
    const timerBtn = $('#rmw-timer');
    const timerMenu = document.createElement('div');
    timerMenu.id = 'rmw-timer-menu';
    document.body.appendChild(timerMenu);

    let sleepEnd = 0;
    let sleepMinutes = 0;
    let sleepTimeout = null;
    let sleepTick = null;

    function formatRemaining(ms) {
        const seconds = Math.max(0, Math.ceil(ms / 1000));
        return seconds >= 60 ? Math.ceil(seconds / 60) + 'm' : seconds + 's';
    }

    function updateTimerButton() {
        if (!sleepEnd) {
            timerBtn.textContent = '⏱';
            timerBtn.classList.remove('rmw-on');
            timerBtn.title = 'Temporizador de apagado';
            return;
        }
        const remaining = formatRemaining(sleepEnd - Date.now());
        timerBtn.textContent = remaining;
        timerBtn.classList.add('rmw-on');
        timerBtn.title = 'Se apagará en ' + remaining;
    }

    function cancelSleepTimer() {
        clearTimeout(sleepTimeout);
        clearInterval(sleepTick);
        sleepTimeout = null;
        sleepTick = null;
        sleepEnd = 0;
        sleepMinutes = 0;
        updateTimerButton();
    }

    function finishSleepTimer() {
        cancelSleepTimer();
        userPaused = true;
        audio.pause();
    }

    function startSleepTimer(minutes) {
        cancelSleepTimer();
        sleepMinutes = minutes;
        sleepEnd = Date.now() + minutes * 60000;
        sleepTimeout = setTimeout(finishSleepTimer, minutes * 60000);
        sleepTick = setInterval(() => {
            if (Date.now() >= sleepEnd) finishSleepTimer();
            else updateTimerButton();
        }, 1000);
        updateTimerButton();
    }

    function renderTimerMenu() {
        timerMenu.innerHTML =
            '<div class="rmw-timer-title">Apagar la radio en…</div>' +
            TIMER_OPTIONS.map(min =>
                `<button type="button" data-min="${min}" class="${sleepMinutes === min ? 'rmw-sel' : ''}">${min} minutos</button>`
            ).join('') +
            (sleepEnd ? '<button type="button" data-min="0" class="rmw-off">Cancelar temporizador</button>' : '');
    }

    function closeTimerMenu() {
        timerMenu.classList.remove('rmw-visible');
    }

    function openTimerMenu() {
        renderTimerMenu();
        const rect = timerBtn.getBoundingClientRect();
        const root = document.documentElement;
        timerMenu.style.right = Math.max(8, root.clientWidth - rect.right) + 'px';
        timerMenu.style.bottom = (root.clientHeight - rect.top + 8) + 'px';
        timerMenu.classList.add('rmw-visible');
    }

    timerBtn.addEventListener('click', () => {
        if (timerMenu.classList.contains('rmw-visible')) closeTimerMenu();
        else openTimerMenu();
    });

    timerMenu.addEventListener('click', event => {
        const option = event.target.closest('button[data-min]');
        if (!option) return;
        const minutes = Number(option.getAttribute('data-min'));
        if (minutes > 0) startSleepTimer(minutes);
        else cancelSleepTimer();
        closeTimerMenu();
    });

    document.addEventListener('click', event => {
        if (!timerMenu.classList.contains('rmw-visible')) return;
        if (timerMenu.contains(event.target) || timerBtn.contains(event.target)) return;
        closeTimerMenu();
    });

    favButton.addEventListener('click', () => { showingFavorites = true; updateModeButtons(); renderFavorites(); });

        searchTimer = setTimeout(() => {

            else loadStations(true);
        }, 350);
    }

    searchInput.addEventListener('input', filtersChanged);
    countrySelect.addEventListener('change', filtersChanged);
    genreSelect.addEventListener('change', filtersChanged);

    $('#rmw-refresh').addEventListener('click', async () => {
        if (showingFavorites) renderFavorites();
        else await loadStations(true);
        loadCountries();
    });
